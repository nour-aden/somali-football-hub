      /* ---- player.js ---- */ /* Macalin Academy: lesson player Walks a student through a lesson: reading pages, then exercises (instant feedback), then a scored test, then a results page. Back and Next move between steps. Progress (best test score per lesson) is kept in this browser only. */ (function () { "use strict"; var DATA = window.MACALIN; var CONTENT = window.MACALIN_CONTENT; var dlg = null; var st = null; var progress = {}; try { progress = JSON.parse(localStorage.getItem("macalin-progress") || "{}") || {}; } catch (e) { progress = {}; } function saveProgress() { try { localStorage.setItem("macalin-progress", JSON.stringify(progress)); } catch (e) { /* storage can be blocked */ } } function esc(s) { return String(s).replace(/[&<>"']/g, function (c) { return { "&": "&", "<": "<", ">": ">", '"': """, "'": "'" }[c]; }); } function letter(i) { return "ABCD".charAt(i); } function subjectOf(id) { return DATA.subjects.filter(function (s) { return s.id === id; })[0]; } var ICON = { back: '', next: '', close: '', ok: '', no: '' }; var PHASE = { learn: "Lesson", practice: "Exercise", test: "Test" }; function buildSteps(c) { var steps = []; c.pages.forEach(function (p, i) { steps.push({ type: "learn", n: i + 1, of: c.pages.length, page: p, key: "learn" + i }); }); c.exercises.forEach(function (q, i) { steps.push({ type: "practice", n: i + 1, of: c.exercises.length, q: q, key: "practice" + i }); }); c.test.forEach(function (q, i) { steps.push({ type: "test", n: i + 1, of: c.test.length, q: q, key: "test" + i }); }); steps.push({ type: "result", key: "result" }); return steps; } function counts(id) { var c = CONTENT[id]; return c ? { pages: c.pages.length, exercises: c.exercises.length, test: c.test.length } : null; } /* ---------- step bodies ---------- */ function learnBody(step) { var p = step.page, html = '

      ' + esc(p.title) + "

      "; p.text.forEach(function (t) { html += "

      " + esc(t) + "

      "; }); if (p.example) { html += '

      ' + esc(p.example.label) + "

        " + p.example.lines.map(function (l) { return "
      • " + esc(l) + "
        • "; }).join("") + "
      "; } if (p.key) html += '

      Remember: ' + esc(p.key) + "

      "; return html; } function optionsHtml(step, reveal) { var chosen = st.ans[step.key]; return '
      ' + step.q.options.map(function (o, k) { var cls = "pl-opt"; if (reveal && chosen !== undefined) { if (k === step.q.answer) cls += " right"; else if (k === chosen) cls += " wrong"; } else if (chosen === k) cls += " picked"; var locked = reveal && chosen !== undefined; return '"; }).join("") + "
      "; } function questionBody(step) { var reveal = step.type === "practice"; var html = '

      ' + esc(step.q.q) + "

      " + optionsHtml(step, reveal); var chosen = st.ans[step.key]; if (reveal && chosen !== undefined) { var good = chosen === step.q.answer; html += '
      ' + (good ? ICON.ok : ICON.no) + "
      " + (good ? "Correct!" : "Not quite.") + "

      " + (good ? "" : "The right answer is " + letter(step.q.answer) + ": " + esc(step.q.options[step.q.answer]) + ". ") + esc(step.q.why) + "

      "; } if (!reveal) html += '

      You will see your score and the right answers at the end.

      '; return html; } function score() { var right = 0; st.c.test.forEach(function (q, i) { if (st.ans["test" + i] === q.answer) right++; }); return right; } function practiceScore() { var right = 0; st.c.exercises.forEach(function (q, i) { if (st.ans["practice" + i] === q.answer) right++; }); return right; } function resultBody() { var total = st.c.test.length, right = score(); var pct = Math.round(right / total * 100); var msg = pct >= 75 ? "Great work! You understand this lesson well." : pct >= 50 ? "Good effort. Read the lesson again and try the test once more." : "Keep going. Read the lesson again, then retry the test."; var html = '
      ' + right + "/" + total + "" + pct + "%
      " + '

      Test complete

      ' + msg + "

      " + '

      Exercises: ' + practiceScore() + " of " + st.c.exercises.length + " correct

      "; html += '

      Review your answers

        '; st.c.test.forEach(function (q, i) { var a = st.ans["test" + i], good = a === q.answer; html += '
      1. ' + (good ? ICON.ok : ICON.no) + "
        " + esc(q.q) + "" + "

        Your answer: " + (a === undefined ? "none" : esc(q.options[a])) + "

        " + (good ? "" : "

        Right answer: " + esc(q.options[q.answer]) + "

        ") + '

        ' + esc(q.why) + "

        2. "; }); html += "
      "; return html; } /* ---------- render ---------- */ function render(keepScroll) { var step = st.steps[st.i], l = st.lesson, last = st.steps.length - 1; var oldBody = dlg.querySelector(".pl-body"); var scrollTop = keepScroll && oldBody ? oldBody.scrollTop : 0; dlg.setAttribute("data-s", l.subject); var segs = st.steps.slice(0, last).map(function (x, k) { return ''; }).join(""); var badge = step.type === "result" ? "Results" : PHASE[step.type] + " " + step.n + " of " + step.of; var body = step.type === "learn" ? learnBody(step) : step.type === "result" ? resultBody() : questionBody(step); var needsAnswer = (step.type === "practice" || step.type === "test") && st.ans[step.key] === undefined; var nextLabel = "Next"; if (step.type === "learn" && st.i === 0) nextLabel = "Start"; if (step.type === "learn" && st.steps[st.i + 1].type === "practice") nextLabel = "Go to exercises"; if (step.type === "practice" && st.steps[st.i + 1].type === "test") nextLabel = "Go to the test"; if (step.type === "test" && st.steps[st.i + 1].type === "result") nextLabel = "Finish test"; var footer = step.type === "result" ? '' + 'Done' + '' + '' : '" + 'Step ' + (st.i + 1) + " of " + last + "" + '"; dlg.innerHTML = '
      ' + esc(l.title) + '' + '
      " + '
      ' + segs + "
      " + '

      ' + esc(badge) + "

      " + body + "
      " + '
      ' + footer + "
      "; var nb = dlg.querySelector(".pl-body"); nb.scrollTop = scrollTop; if (step.type === "result") { var right = score(), total = st.c.test.length, prev = progress[l.id]; progress[l.id] = { best: Math.max(prev ? prev.best : 0, right), total: total }; saveProgress(); // Send each finished attempt to Supabase once; a failure never interrupts the student. var signature = JSON.stringify(st.c.test.map(function (q, i) { return st.ans["test" + i]; })); if (st.sent !== signature && window.MacalinApi && window.MacalinApi.enabled) { st.sent = signature; window.MacalinApi.saveResult({ lessonId: l.id, score: right, total: total }).catch(function () { /* offline or table missing */ }); } } } function go(delta) { var next = st.i + delta; if (next < 0 || next >= st.steps.length) return; st.i = next; render(false); var nb = dlg.querySelector(".pl-body"); if (nb) nb.focus({ preventScroll: true }); } /* ---------- open / close ---------- */ function open(id) { var lesson = DATA.lessons.filter(function (l) { return l.id === id; })[0]; var c = CONTENT[id]; if (!lesson || !c) return false; dlg = document.getElementById("player"); st = { lesson: lesson, c: c, steps: buildSteps(c), i: 0, ans: {} }; render(false); if (dlg.showModal) { if (!dlg.open) dlg.showModal(); } else { dlg.setAttribute("open", ""); } return true; } function exit() { if (!dlg) return; if (dlg.close) dlg.close(); else dlg.removeAttribute("open"); } function init() { dlg = document.getElementById("player"); if (!dlg) return; dlg.addEventListener("click", function (e) { if (!st) return; var t = e.target, el; if (t.closest("[data-pl-exit]")) { exit(); return; } if (t.closest("[data-pl-back]")) { go(-1); return; } if (t.closest("[data-pl-next]")) { if (!t.closest("[data-pl-next]").disabled) go(1); return; } if (t.closest("[data-pl-retry]")) { for (var k in st.ans) { if (k.indexOf("test") === 0) delete st.ans[k]; } st.i = st.steps.findIndex(function (s) { return s.type === "test"; }); render(false); return; } if ((el = t.closest("[data-opt]"))) { var step = st.steps[st.i]; if (step.type === "practice" && st.ans[step.key] !== undefined) return; st.ans[step.key] = Number(el.getAttribute("data-opt")); render(true); var picked = dlg.querySelector(".pl-opt.picked, .pl-opt.right, .pl-opt.wrong"); if (picked && step.type === "test") picked.focus({ preventScroll: true }); } }); dlg.addEventListener("keydown", function (e) { if (!st) return; if (e.key === "ArrowRight") { var nx = dlg.querySelector("[data-pl-next]"); if (nx && !nx.disabled) { e.preventDefault(); go(1); } } else if (e.key === "ArrowLeft") { if (st.i > 0) { e.preventDefault(); go(-1); } } }); dlg.addEventListener("close", function () { window.dispatchEvent(new Event("macalin:progress")); }); } window.MacalinPlayer = { open: open, counts: counts, best: function (id) { return progress[id] || null; } }; init(); })();
